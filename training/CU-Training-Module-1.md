import struct  
import hmac

class MockQuicStream:  
    """Simulates a secure network connection stream delivering binary data frames."""  
    def \_\_init\_\_(self, data: bytes):  
        self.buffer \= data  
        self.ptr \= 0  
        self.closed \= False

    def read\_exact(self, n: int) \-\> bytes:  
        if self.closed:  
            raise ConnectionError("Stream closed.")  
        if self.ptr \+ n \> len(self.buffer):  
            raise ValueError("Unexpected End of Stream.")  
        chunk \= self.buffer\[self.ptr:self.ptr+n\]  
        self.ptr \+= n  
        return chunk

    def close\_with\_error(self, error\_code: int):  
        self.closed \= True  
        self.error\_code \= error\_code

        Enforces fixed cryptographic framing rules. Drops connections on violations.  
        """  
        try:  
            \# 1\. Read Fixed Envelope Header Length (4 bytes magic code \+ 32 bytes pubkey \+ 8 bytes seq \+ 2 bytes length)

            magic, pubkey\_bytes, seq, payload\_len \= struct.unpack("\>4s32sQH", envelope\_bytes)

            if magic \!= b"CUFB":  
                stream.close\_with\_error(cls.PROTOCOL\_ERROR\_CODE)  
                raise ConnectionError("Malformed frame protocol signature.")

            if payload\_len \> cls.MAX\_PAYLOAD\_SIZE:  
                stream.close\_with\_error(cls.PROTOCOL\_ERROR\_CODE)  
                raise ConnectionError("Payload length exceeds allowed strict cluster MTU boundaries.")

            \# 2\. Extract Signature Envelope (e.g., 64 bytes Ed25519)  
            signature\_bytes \= stream.read\_exact(64)

            \# 3\. Stream Payload Memory Allocation  
            payload\_bytes \= stream.read\_exact(payload\_len)

            \# 4\. Cryptographic Validation Gate  
            if pubkey\_bytes not in registered\_nodes:  
                stream.close\_with\_error(cls.PROTOCOL\_ERROR\_CODE)

            if not is\_valid\_sig:
